/* ========================================== Entreprise ARASHI v4.0 - ERP Dashboard Fichier : js/erp.js ========================================== */ import { supabase } from './supabase.js'; /** * Initialise le tableau de bord ERP */ export async function loadErpDashboard() { initNavigationRouting(); await Promise.all([ loadStocksAndLogistics(), loadConstructionProjects(), loadCrmData() ]); } /** * Prévient les erreurs 404 en vérifiant l'existence des routes de l'ERP */ function initNavigationRouting() { const navLinks = document.querySelectorAll("aside.sidebar nav a"); navLinks.forEach(link => { link.addEventListener("click", (e) => { const href = link.getAttribute("href"); if (!href || href === "#") { e.preventDefault(); console.warn("Lien non défini ou en cours de développement."); } }); }); } /** * 📦 1. STOCKS & LOGISTIQUE * Suivi du matériel BTP et alerte sur seuil critique */ async function loadStocksAndLogistics() { const tableBody = document.getElementById("stocksTableBody"); const alertBadge = document.getElementById("stockAlertBadge"); try { const { data: inventory, error } = await supabase .from("inventory_equipment") .select("*") .order("quantity", { ascending: true }); if (error) throw error; let lowStockCount = 0; if (tableBody) { tableBody.innerHTML = ""; if (!inventory || inventory.length === 0) { tableBody.innerHTML = ` Aucun équipement répertorié. `; } else { inventory.forEach(item => { const isLowStock = item.quantity <= (item.min_threshold || 5); if (isLowStock) lowStockCount++; const row = document.createElement("tr"); row.style.borderBottom = "1px solid var(--border)"; row.innerHTML = ` ${item.name || 'Matériel'} ${item.category || 'BTP'} ${item.quantity} ${item.unit || 'unités'} ${isLowStock ? '⚠️ Seuil Critique' : 'OK'} `; tableBody.appendChild(row); }); } } if (alertBadge) { alertBadge.textContent = `${lowStockCount} alerte(s)`; alertBadge.className = `badge ${lowStockCount > 0 ? 'badge-danger' : 'badge-success'}`; } } catch (err) { console.error("Erreur chargement stocks BTP :", err); } } /** * 🏗️ 2. CHANTIERS & PROJETS * Supervision des chantiers et taux d'avancement */ async function loadConstructionProjects() { const projectsContainer = document.getElementById("projectsSupervisionGrid"); try { const { data: projects, error } = await supabase .from("construction_projects") .select("*") .order("created_at", { ascending: false }); if (error) throw error; if (projectsContainer) { projectsContainer.innerHTML = ""; if (!projects || projects.length === 0) { projectsContainer.innerHTML = `
Aucun chantier en cours de supervision.
`; return; } projects.forEach(project => { const progress = project.progress_percent || 0; const card = document.createElement("div"); card.className = "card"; card.style.cssText = "padding: 15px; border-left: 4px solid var(--primary-color);"; card.innerHTML = `

${project.project_type || 'Chantier BTP'}

${project.status || 'En cours'}

📍 ${project.location || 'Site non spécifié'}

Avancement ${progress}%
`; projectsContainer.appendChild(card); }); } } catch (err) { console.error("Erreur chargement suivi des chantiers :", err); } } /** * 🤝 3. RELATION CLIENT (CRM) * Opportunités et devis d'affaires */ async function loadCrmData() { const crmTable = document.getElementById("crmOpportunitiesTable"); try { const { data: leads, error } = await supabase .from("crm_opportunities") .select("*") .order("created_at", { ascending: false }); if (error) throw error; if (crmTable) { crmTable.innerHTML = ""; if (!leads || leads.length === 0) { crmTable.innerHTML = ` Aucune opportunité CRM enregistrée. `; return; } leads.forEach(lead => { const row = document.createElement("tr"); row.style.borderBottom = "1px solid var(--border)"; row.innerHTML = ` ${lead.client_name || 'Client Inconnu'} ${lead.subject || 'Devis BTP'} ${lead.estimated_value_pi || 0} π ${lead.status || 'En négociation'} `; crmTable.appendChild(row); }); } } catch (err) { console.error("Erreur chargement CRM :", err); } } document.addEventListener("DOMContentLoaded", () => { loadErpDashboard(); });
